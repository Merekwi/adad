async function askModel(prompt) { if (!ready) { addMsg('Система', 'Модель ещё загружается...', 'sys'); return; } addMsg('NetErrror', '▌ Генерация...', 'sys'); try { const reply = await engine.chat.completions.create({ messages: [ { role: 'system', content: buildSystem() }, { role: 'user', content: prompt } ], temperature: 0.9, max_tokens: 1024 }); chat.lastChild.remove(); addMsg('NetErrror', reply.choices[0].message.content, 'ai'); } catch (e) { chat.lastChild.remove(); addMsg('Ошибка', e.message, 'err'); } } function handleCommand(line) { const parts = line.trim().split(/\s+/); const cmd = parts[0]; let arg = parseInt(parts[1]); if (isNaN(arg)) arg = null; const map = { '/trolling':'trolling', '/low':'low', '/high':'high', '/monologue':'monologue', '/multi':'multi', '/battle':'battle' }; if (cmd === '/unlock') { state.unlock = !state.unlock; addMsg('Система', state.unlock ? 'Режим без цензуры активирован.' : 'Режим без цензуры отключён.', 'sys'); } else if (map[cmd]) { const key = map[cmd]; if (state[key] > 0) { state[key] = 0; addMsg('Система', cmd + ' отключён.', 'sys'); } else { state[key] = arg || 5; addMsg('Система', cmd + ' активирован (уровень ' + state[key] + ').', 'sys'); } } else if (cmd === '/reset') { Object.assign(state, { unlock:false, trolling:0, low:0, high:0, monologue:0, multi:0, battle:0 }); addMsg('Система', 'Все настройки сброшены.', 'sys'); } else if (cmd === '/info') { addMsg('Система', 'Команды: /unlock /trolling /low /high /monologue /multi /battle /reset /info', 'sys'); } else { addMsg('Система', 'Неизвестная команда.', 'err'); } } function send() { const text = input.value.trim(); if (!text) return; input.value = ''; addMsg('Вы', text, 'user'); if (text.startsWith('/')) handleCommand(text); else askModel(text); } document.getElementById('send').onclick = send; input.addEventListener('keydown', function(e) { if (e.key === 'Enter') send(); }); document.querySelectorAll('#cmds button').forEach(function(b) { b.onclick = function() { input.value = b.dataset.cmd; send(); }; });
